# ─── CARGA DE DATOS ───────────────────────────────────
def cargar_csv():
    ruta = input(f"{Fore.WHITE}Ruta del CSV (o Enter para demo): {Fore.GREEN}").strip()

    if not ruta:
        print(f"{Fore.YELLOW}[INFO] Cargando dataset demo (iris)...")
        from sklearn.datasets import load_iris
        iris = load_iris(as_frame=True)
        df = iris.frame
        df.columns = [c.replace(" (cm)", "").replace(" ", "_") for c in df.columns]
        print(f"{Fore.GREEN}[OK] Dataset demo cargado: {df.shape[0]} filas × {df.shape[1]} columnas")
        return df

    if not os.path.exists(ruta):
        print(f"{Fore.RED}[ERROR] Archivo no encontrado.")
        return None

    try:
        sep = input(f"{Fore.WHITE}Separador (Enter=coma, escribe ';' o '\\t'): {Fore.GREEN}").strip() or ","
        df = pd.read_csv(ruta, sep=sep)
        print(f"{Fore.GREEN}[OK] Cargado: {df.shape[0]} filas × {df.shape[1]} columnas")
        return df
    except Exception as e:
        print(f"{Fore.RED}[ERROR] {e}")
        return None

# ─── VALIDACIÓN DE DATASET ───────────────────────────
def validar_dataset(df):
    """
    Valida que el DataFrame cargado sea apto para el análisis.
    Retorna (es_valido: bool, advertencias: list, errores: list)
    """
    errores = []
    advertencias = []

    separador("VALIDACIÓN DEL DATASET")

    # ── 1. DIMENSIONES ──────────────────────────────
    filas, cols = df.shape
    print(f"\n{Fore.WHITE}{'─'*54}")
    print(f"  {Fore.CYAN}Dimensiones:{Fore.WHITE} {filas} filas × {cols} columnas")

    if filas < 10:
        errores.append(f"Muy pocas filas ({filas}). Mínimo recomendado: 30.")
    elif filas < 30:
        advertencias.append(f"Pocas filas ({filas}). Los modelos pueden no generalizar bien.")

    if cols < 2:
        errores.append("Se necesitan al menos 2 columnas (1 feature + 1 target).")

    # ── 2. VALORES NULOS ────────────────────────────
    nulos = df.isnull().sum()
    pct_nulos = (nulos / filas * 100).round(2)
    cols_con_nulos = nulos[nulos > 0]

    if len(cols_con_nulos) > 0:
        print(f"\n  {Fore.YELLOW}⚠ Columnas con valores nulos:")
        for col, n in cols_con_nulos.items():
            pct = pct_nulos[col]
            color = Fore.RED if pct > 40 else Fore.YELLOW
            print(f"    {color}{col:<30} {n:>5} nulos  ({pct:.1f}%)")
            if pct > 40:
                advertencias.append(f"'{col}' tiene {pct:.1f}% de nulos — considerar eliminar.")
    else:
        print(f"  {Fore.GREEN}✓ Sin valores nulos")

    # ── 3. TIPOS DE DATOS ───────────────────────────
    tipos = df.dtypes.value_counts()
    n_numericas  = len(df.select_dtypes(include=["number"]).columns)
    n_categoricas = len(df.select_dtypes(include=["object", "category"]).columns)

    print(f"\n  {Fore.CYAN}Tipos de columnas:")
    print(f"    {Fore.WHITE}Numéricas:    {Fore.GREEN}{n_numericas}")
    print(f"    {Fore.WHITE}Categóricas:  {Fore.YELLOW}{n_categoricas}")

    if n_numericas == 0:
        advertencias.append("No hay columnas numéricas. Correlación no será útil.")

    # ── 4. DUPLICADOS ───────────────────────────────
    n_dup = df.duplicated().sum()
    if n_dup > 0:
        pct_dup = round(n_dup / filas * 100, 2)
        color = Fore.RED if pct_dup > 20 else Fore.YELLOW
        print(f"\n  {color}⚠ Filas duplicadas: {n_dup} ({pct_dup:.1f}%)")
        if pct_dup > 20:
            advertencias.append(f"{pct_dup:.1f}% de filas duplicadas — puede sesgar resultados.")
    else:
        print(f"  {Fore.GREEN}✓ Sin duplicados")

    # ── 5. COLUMNAS CON UN SOLO VALOR (sin varianza) ─
    sin_varianza = [c for c in df.columns if df[c].nunique() <= 1]
    if sin_varianza:
        print(f"\n  {Fore.RED}✗ Columnas sin varianza (valor constante):")
        for c in sin_varianza:
            print(f"    {Fore.RED}{c}")
        errores.append(f"Columnas constantes detectadas: {sin_varianza} — no aportan información.")

    # ── 6. CARDINALIDAD ALTA EN CATEGÓRICAS ─────────
    cat_cols = df.select_dtypes(include=["object", "category"]).columns
    for col in cat_cols:
        uniq = df[col].nunique()
        if uniq > 50:
            advertencias.append(f"'{col}' tiene {uniq} categorías únicas — alta cardinalidad.")
            print(f"\n  {Fore.YELLOW}⚠ '{col}' tiene {uniq} categorías únicas (alta cardinalidad)")

    # ── 7. DESBALANCE EN TARGET (se revisa al elegir target) ─
    # (se evalúa más tarde cuando se conoce el target)

    # ── RESUMEN FINAL ───────────────────────────────
    print(f"\n{Fore.CYAN}{'─'*54}")
    if errores:
        print(f"  {Fore.RED}✗ ERRORES CRÍTICOS ({len(errores)}):")
        for e in errores:
            print(f"    {Fore.RED}• {e}")

    if advertencias:
        print(f"  {Fore.YELLOW}⚠ ADVERTENCIAS ({len(advertencias)}):")
        for a in advertencias:
            print(f"    {Fore.YELLOW}• {a}")

    if not errores and not advertencias:
        print(f"  {Fore.GREEN}✓ Dataset válido. Sin problemas detectados.")
    elif not errores:
        print(f"\n  {Fore.GREEN}✓ Dataset APTO para análisis (con advertencias).")
    else:
        print(f"\n  {Fore.RED}✗ Dataset NO APTO. Corrige los errores antes de continuar.")

    print(f"{Fore.CYAN}{'─'*54}")
    return len(errores) == 0, advertencias, errores


def validar_target(df, target_col):
    """Validaciones específicas una vez que se conoce el target."""
    advertencias = []
    separador(f"VALIDACIÓN DEL TARGET: '{target_col}'")

    y = df[target_col]
    n_unique = y.nunique()
    es_categorico = y.dtype == object or str(y.dtype) == "category"

    print(f"\n  {Fore.CYAN}Tipo:{Fore.WHITE} {'Categórico' if es_categorico else 'Numérico'}")
    print(f"  {Fore.CYAN}Valores únicos:{Fore.WHITE} {n_unique}")

    if es_categorico or n_unique <= 20:
        # Revisar desbalance de clases
        conteo = y.value_counts()
        total = len(y)
        print(f"\n  {Fore.WHITE}Distribución de clases:")
        for clase, cnt in conteo.items():
            pct = cnt / total * 100
            bar = "█" * int(pct / 3)
            color = Fore.GREEN if pct > 30 else Fore.YELLOW if pct > 10 else Fore.RED
            print(f"    {color}{str(clase):<20} {cnt:>5} ({pct:5.1f}%)  {Fore.CYAN}{bar}")

        ratio = conteo.max() / conteo.min() if conteo.min() > 0 else float("inf")
        if ratio > 10:
            advertencias.append(f"Desbalance severo (ratio {ratio:.1f}:1). Considera técnicas de balanceo.")
            print(f"\n  {Fore.RED}⚠ Desbalance severo detectado (ratio {ratio:.1f}:1)")
        elif ratio > 3:
            advertencias.append(f"Desbalance moderado (ratio {ratio:.1f}:1).")
            print(f"\n  {Fore.YELLOW}⚠ Desbalance moderado (ratio {ratio:.1f}:1)")
        else:
            print(f"\n  {Fore.GREEN}✓ Clases balanceadas")
    else:
        # Numérico continuo: revisar distribución básica
        print(f"\n  {Fore.WHITE}Estadísticas básicas:")
        print(f"    {Fore.CYAN}Media:   {Fore.WHITE}{y.mean():.4f}")
        print(f"    {Fore.CYAN}Mediana: {Fore.WHITE}{y.median():.4f}")
        print(f"    {Fore.CYAN}Std:     {Fore.WHITE}{y.std():.4f}")
        print(f"    {Fore.CYAN}Min/Max: {Fore.WHITE}{y.min():.4f} / {y.max():.4f}")

    if advertencias:
        print(f"\n  {Fore.YELLOW}⚠ Advertencias del target:")
        for a in advertencias:
            print(f"    • {a}")

    print(f"{Fore.CYAN}{'─'*54}")


def elegir_target(df):
    separador("Columnas disponibles")
    for i, col in enumerate(df.columns):
        dtype = str(df[col].dtype)
        print(f"  {Fore.WHITE}[{i:2d}]{Fore.YELLOW} {col:<30} {Fore.CYAN}{dtype}")
    idx = input(f"\n{Fore.WHITE}Número de la columna TARGET: {Fore.GREEN}").strip()
    try:
        target_col = df.columns[int(idx)]
        print(f"{Fore.GREEN}[OK] Target: '{target_col}'")
        return target_col
    except:
        print(f"{Fore.RED}[ERROR] Índice inválido.")
        return None

def preparar_datos(df, target_col):
    from sklearn.preprocessing import LabelEncoder
    le = LabelEncoder()

    X = df.drop(columns=[target_col]).copy()
    y = df[target_col].copy()

    # Codificar target si es categórico
    if y.dtype == object or str(y.dtype) == "category":
        y = le.fit_transform(y.astype(str))

    # Codificar features categóricas
    for col in X.select_dtypes(include=["object", "category"]).columns:
        X[col] = le.fit_transform(X[col].astype(str))

    # Rellenar nulos
    X = X.fillna(X.median(numeric_only=True))
    X = X.fillna(0)

    return X, y

# ─── MÉTODO 1: CORRELACIÓN ────────────────────────────
def metodo_correlacion(df, target_col):
    separador("MÉTODO 1 — Correlación con el target")

    X, y = preparar_datos(df, target_col)
    correlaciones = X.corrwith(pd.Series(y, name=target_col)).abs().sort_values(ascending=False)

    print(f"\n{Fore.WHITE}{'Variable':<35} {'Correlación |r|':>15}")
    print(f"{Fore.CYAN}{'─'*52}")
    for var, val in correlaciones.items():
        bar = "█" * int(val * 30)
        color = Fore.GREEN if val > 0.5 else Fore.YELLOW if val > 0.3 else Fore.WHITE
        print(f"  {color}{var:<33} {val:>8.4f}  {Fore.CYAN}{bar}")

    return correlaciones

# ─── MÉTODO 2: CHI² ───────────────────────────────────
def metodo_chi2(df, target_col):
    separador("MÉTODO 2 — Chi² (SelectKBest)")
    from sklearn.feature_selection import SelectKBest, f_classif
    from sklearn.preprocessing import MinMaxScaler

    X, y = preparar_datos(df, target_col)

    # Escalar para chi2 (valores >= 0)
    scaler = MinMaxScaler()
    X_scaled = scaler.fit_transform(X)

    selector = SelectKBest(score_func=f_classif, k="all")
    selector.fit(X_scaled, y)

    scores = pd.Series(selector.scores_, index=X.columns).sort_values(ascending=False)
    scores_norm = (scores / scores.max())  # normalizar para visualización

    print(f"\n{Fore.WHITE}{'Variable':<35} {'Score F':>12}  {'Importancia relativa'}")
    print(f"{Fore.CYAN}{'─'*70}")
    for var, val in scores.items():
        rel = scores_norm[var]
        bar = "█" * int(rel * 25)
        color = Fore.GREEN if rel > 0.7 else Fore.YELLOW if rel > 0.4 else Fore.WHITE
        print(f"  {color}{var:<33} {val:>10.2f}  {Fore.CYAN}{bar}")

    return scores

# ─── MÉTODO 3: RANDOM FOREST ─────────────────────────
def metodo_random_forest(df, target_col):
    separador("MÉTODO 3 — Random Forest (importancia de características)")
    from sklearn.ensemble import RandomForestClassifier, RandomForestRegressor

    X, y = preparar_datos(df, target_col)

    # Decidir si clasificación o regresión
    n_unique = len(np.unique(y))
    if n_unique <= 20 or df[target_col].dtype == object:
        modelo = RandomForestClassifier(n_estimators=100, random_state=42)
        tipo = "Clasificación"
    else:
        modelo = RandomForestRegressor(n_estimators=100, random_state=42)
        tipo = "Regresión"

    print(f"{Fore.YELLOW}[INFO] Modo: {tipo} | Entrenando Random Forest...")
    modelo.fit(X, y)

    importancias = pd.Series(modelo.feature_importances_, index=X.columns).sort_values(ascending=False)

    print(f"\n{Fore.WHITE}{'Variable':<35} {'Importancia':>12}  {'Peso visual'}")
    print(f"{Fore.CYAN}{'─'*70}")
    for var, val in importancias.items():
        bar = "█" * int(val * 100)
        color = Fore.GREEN if val > 0.15 else Fore.YELLOW if val > 0.05 else Fore.WHITE
        print(f"  {color}{var:<33} {val:>10.4f}  {Fore.CYAN}{bar}")

    return importancias

# ─── MÉTODO COMPARATIVO ───────────────────────────────
def comparar_metodos(df, target_col):
    separador("COMPARACIÓN DE LOS 3 MÉTODOS")

    corr = metodo_correlacion(df, target_col)
    chi  = metodo_chi2(df, target_col)
    rf   = metodo_random_forest(df, target_col)

    # Normalizar todos a 0-1
    def norm(s):
        return (s - s.min()) / (s.max() - s.min() + 1e-9)

    resumen = pd.DataFrame({
        "Correlacion": norm(corr),
        "Chi2_F":      norm(chi),
        "RandomForest": norm(rf),
    }).fillna(0)

    resumen["SCORE_FINAL"] = resumen.mean(axis=1)
    resumen = resumen.sort_values("SCORE_FINAL", ascending=False)

    separador("RANKING FINAL (promedio normalizado de los 3 métodos)")
    print(f"\n{Fore.WHITE}{'Variable':<35} {'Corr':>7} {'Chi²':>7} {'RF':>7} {'FINAL':>7}  {'Ranking visual'}")
    print(f"{Fore.CYAN}{'─'*85}")

    for i, (var, row) in enumerate(resumen.iterrows(), 1):
        bar = "█" * int(row["SCORE_FINAL"] * 30)
        medal = {1: "🥇", 2: "🥈", 3: "🥉"}.get(i, f"#{i:2d}")
        color = Fore.GREEN if i <= 3 else Fore.YELLOW if i <= 6 else Fore.WHITE
        print(f"  {color}{medal} {var:<32} {row['Correlacion']:>7.3f} {row['Chi2_F']:>7.3f} {row['RandomForest']:>7.3f} {row['SCORE_FINAL']:>7.3f}  {Fore.CYAN}{bar}")

    return resumen

# ─── EXPORTAR ─────────────────────────────────────────
def exportar_resultados(resumen, target_col):
    nombre = input(f"\n{Fore.WHITE}Nombre del archivo (sin .csv): {Fore.GREEN}").strip() or "resultados_seleccion"
    ruta_out = f"{nombre}.csv"
    resumen.to_csv(ruta_out)
    print(f"{Fore.GREEN}[OK] Resultados exportados a: {ruta_out}")

# ─── MAIN LOOP ────────────────────────────────────────
def main():
    check_and_import()
    banner()

    df = None
    target_col = None
    ultimo_resumen = None

    while True:
        opcion = menu_principal()

        if opcion == "0":
            print(f"\n{Fore.CYAN}Hasta luego.\n")
            break

        elif opcion == "1":
            df = cargar_csv()
            if df is not None:
                print(f"\n{Fore.CYAN}Vista previa:")
                print(df.head(3).to_string())
                # Validación del dataset
                es_valido, advertencias, errores = validar_dataset(df)
                if not es_valido:
                    continuar = input(f"\n{Fore.RED}¿Continuar de todas formas? (s/N): {Fore.GREEN}").strip().lower()
                    if continuar != "s":
                        df = None
                        print(f"{Fore.YELLOW}[INFO] Carga cancelada. Corrige el dataset e intenta de nuevo.")
                        input(f"\n{Fore.CYAN}[Enter para continuar...]")
                        continue
                target_col = elegir_target(df)
                if target_col:
                    validar_target(df, target_col)

        elif opcion in ["2", "3", "4", "5"]:
            if df is None or target_col is None:
                print(f"{Fore.RED}[ERROR] Primero carga un CSV (opción 1).")
                continue
            if opcion == "2":
                metodo_correlacion(df, target_col)
            elif opcion == "3":
                metodo_chi2(df, target_col)
            elif opcion == "4":
                metodo_random_forest(df, target_col)
            elif opcion == "5":
                ultimo_resumen = comparar_metodos(df, target_col)

        elif opcion == "6":
            if ultimo_resumen is None:
                print(f"{Fore.RED}[ERROR] Primero ejecuta la comparación (opción 5).")
                continue
            exportar_resultados(ultimo_resumen, target_col)

        else:
            print(f"{Fore.RED}[ERROR] Opción no válida.")

        input(f"\n{Fore.CYAN}[Enter para continuar...]")
        os.system("cls" if os.name == "nt" else "clear")
        banner()


if __name__ == "__main__":
    main()
